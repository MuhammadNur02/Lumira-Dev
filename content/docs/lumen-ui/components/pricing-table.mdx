<TypeTable
  type={{

    highlighted: { type: 'string', description: 'Id of the plan that gets the brand fill.' },
    defaultInterval: {
      type: "'month' | 'year'",

      default: "'month'",
    },
    onSelect: {
      type: '(planId: string, interval: string) => void',
      description: 'Called when a plan button is pressed.',
    },
  }}
/>
